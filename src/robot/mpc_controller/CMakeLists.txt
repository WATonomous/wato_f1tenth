cmake_minimum_required(VERSION 3.16)
project(mpc_controller LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(MPC_BUILD_ROS "Build ROS node and install package" ON)
set(MPC_DEPS_ROOT "$ENV{HOME}/.local/opt/wato-mpc" CACHE PATH "Persistent solver dependencies")
set(ACADOS_ROOT "${MPC_DEPS_ROOT}/acados" CACHE PATH "acados source and installed libraries")
set(MPC_PYTHON "${MPC_DEPS_ROOT}/venv/bin/python" CACHE FILEPATH "Python with CasADi/acados_template")
set(MPC_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/config/prototype.json" CACHE FILEPATH "Generated vehicle profile")
set(CMAKE_BUILD_RPATH "${ACADOS_ROOT}/lib")
set(CMAKE_INSTALL_RPATH "${ACADOS_ROOT}/lib")
# acados itself loads HPIPM/BLASFEO, so the dependency path must be transitive.
add_link_options("-Wl,--disable-new-dtags")
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
execute_process(COMMAND ${CMAKE_COMMAND} -E env
  "ACADOS_SOURCE_DIR=${ACADOS_ROOT}" "MPLCONFIGDIR=${CMAKE_CURRENT_BINARY_DIR}/matplotlib"
  ${MPC_PYTHON} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_solver.py
  --config ${MPC_CONFIG} --output ${GENERATED}
  RESULT_VARIABLE GENERATION_STATUS)
if(NOT GENERATION_STATUS EQUAL 0)
  message(FATAL_ERROR "Solver generation failed. Install dependencies; see README.md")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  tools/generate_solver.py ${MPC_CONFIG})
file(GLOB GENERATED_FUNCTIONS "${GENERATED}/mpc_proto_model/*.c"
  "${GENERATED}/mpc_proto_cost/*.c" "${GENERATED}/mpc_proto_constraints/*.c")
add_library(mpc_generated STATIC ${GENERATED}/acados_solver_mpc_proto.c ${GENERATED_FUNCTIONS})
target_include_directories(mpc_generated PUBLIC ${GENERATED} ${ACADOS_ROOT}/include
  ${ACADOS_ROOT}/include/blasfeo/include ${ACADOS_ROOT}/include/hpipm/include)
target_link_libraries(mpc_generated PUBLIC ${ACADOS_ROOT}/lib/libacados.so
  ${ACADOS_ROOT}/lib/libhpipm.so ${ACADOS_ROOT}/lib/libblasfeo.so m)
add_library(mpc_core src/solver.cpp src/scene.cpp src/controller.cpp)
target_include_directories(mpc_core PUBLIC include)
target_link_libraries(mpc_core PUBLIC mpc_generated)
target_compile_options(mpc_core PRIVATE -Wall -Wextra -Wpedantic)

if(MPC_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(local_planning REQUIRED)
  target_link_libraries(mpc_core PUBLIC local_planning::local_planning_core)
  find_package(rclcpp REQUIRED)
  find_package(ackermann_msgs REQUIRED)
  find_package(global_planner REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(tf2 REQUIRED)
  find_package(tf2_ros REQUIRED)
  find_package(tf2_geometry_msgs REQUIRED)
  find_package(visualization_msgs REQUIRED)
  add_library(mpc_visualization src/visualization.cpp)
  target_link_libraries(mpc_visualization PUBLIC mpc_core)
  ament_target_dependencies(mpc_visualization PUBLIC nav_msgs std_msgs geometry_msgs visualization_msgs)
  add_executable(mpc_node src/node.cpp)
  target_link_libraries(mpc_node PRIVATE mpc_visualization)
  ament_target_dependencies(mpc_node PUBLIC rclcpp ackermann_msgs global_planner nav_msgs
    std_msgs geometry_msgs tf2 tf2_ros tf2_geometry_msgs)
  add_executable(fixture_simulator tools/fixture_simulator.cpp)
  target_link_libraries(fixture_simulator PRIVATE mpc_core)
  ament_target_dependencies(fixture_simulator PUBLIC rclcpp ackermann_msgs global_planner nav_msgs std_msgs)
  install(TARGETS mpc_node fixture_simulator DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
  install(PROGRAMS tools/porto_scene.py tools/autodrive_trial.py DESTINATION lib/${PROJECT_NAME})
  install(FILES ../particle_filter/maps/porto.pgm
    ../global_planner/assets/my_map_clean.csv DESTINATION share/${PROJECT_NAME}/maps)
else()
  add_library(mpc_reference ../local_planning/src/reference/raceline_reference.cpp)
  target_include_directories(mpc_reference PUBLIC ../local_planning/include)
  target_link_libraries(mpc_core PUBLIC mpc_reference)
endif()
include(CTest)
add_executable(mpc_benchmark tools/benchmark.cpp)
target_link_libraries(mpc_benchmark PRIVATE mpc_core)
if(MPC_BUILD_ROS)
  install(TARGETS mpc_benchmark DESTINATION lib/${PROJECT_NAME})
endif()
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  include(GoogleTest)
  add_executable(mpc_tests test/test_model.cpp test/test_solver.cpp test/test_scene.cpp test/test_controller.cpp)
  target_link_libraries(mpc_tests PRIVATE mpc_core GTest::gtest_main)
  gtest_discover_tests(mpc_tests)
  if(MPC_BUILD_ROS)
    add_executable(mpc_visualization_tests test/test_visualization.cpp)
    target_link_libraries(mpc_visualization_tests PRIVATE mpc_visualization GTest::gtest_main)
    gtest_discover_tests(mpc_visualization_tests)
  endif()
endif()
if(MPC_BUILD_ROS)
  ament_package()
endif()
